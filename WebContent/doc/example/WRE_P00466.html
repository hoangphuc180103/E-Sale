4구현 예시

4.1모든 노드의 선택 해제하기

TreeView의 'setNodeSet' 메소드를 사용하여 스크립트를 작성합니다.

Example 1.Script

scwin.btn_ex1_onclick = function (e) {
    // 새로운 dataList의 아이디가 dataList1인 경우
    var columnInfoArr = [
        {"label":"라벨ㄱ",   "value":"01", "depth":"1"},
        {"label":"라벨ㄱ-1", "value":"02", "depth":"2"},
        {"label":"라벨ㄱ-2", "value":"03", "depth":"2"},
        {"label":"라벨ㄱ-3", "value":"04", "depth":"2"},
        {"label":"라벨ㄴ",   "value":"05", "depth":"1"},
        {"label":"라벨ㄴ-1", "value":"06", "depth":"2"},
        {"label":"라벨ㄴ-2", "value":"07", "depth":"2"},
        {"label":"라벨ㄴ-3", "value":"08", "depth":"2"}
    ];
    dlt_exam2.setJSON(columnInfoArr);
    trv2.setNodeSet( "data:dlt_exam2" , columnInfoArr );
};
